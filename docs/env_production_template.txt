# Diz ao Django que a conexão original era HTTPS quando o TLS termina no nginx.
# Sem isto, com SECURE_SSL_REDIRECT ligado, o site entra em loop de redirect.
# Exige que o nginx envie: proxy_set_header X-Forwarded-Proto $scheme;

# --------------------------------------------
# Banco de dados (PostgreSQL)
# --------------------------------------------

# Usadas pelo docker-compose para inicializar o container do Postgres.
# POSTGRES_PASSWORD precisa bater com a senha da DATABASE_URL acima; se ficar
# vazia, o container do Postgres não inicializa um volume novo.

POSTGRES_USER=trader_user
